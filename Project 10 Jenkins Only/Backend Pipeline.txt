pipeline {
    agent any

    environment {
        JAVA_HOME = "C:\\Program Files\\Java\\jdk-11.0.15.1"
        MAVEN_HOME = "C:\\ProgramData\\Jenkins\\.jenkins\\tools\\hudson.tasks.Maven_MavenInstallation\\Maven"
        BACKEND_DIR = "${WORKSPACE}\\ORSProject-10"
        JAR_NAME = "ORSProject-10-0.0.1-SNAPSHOT.jar"
        BACKEND_PORT = "8080"
        PATH = "${JAVA_HOME}\\bin;${MAVEN_HOME}\\bin;${env.PATH}"
    }

    stages {

        stage('Cleanup') {
            steps {
                deleteDir()
            }
        }
stage('Clone Repo') {
    steps {
        git(
            url: 'https://github.com/rishabhshri162/Project-10-Jenkins-Backend.git',
            branch: 'main',
        )
    }
}

        stage('Build') {
            steps {
                dir("${BACKEND_DIR}") {
                    bat "mvn clean package -DskipTests"
                }
            }
        }

      stage('Kill Existing App (Port 8080)') {
    steps {
        bat '''
        netstat -ano | findstr :8080 > port.txt
        for /f "tokens=5" %%a in (port.txt) do (
            taskkill /PID %%a /F
        )
        exit /B 0
        '''
    }
}

        stage('Run') {
            steps {
                dir("${BACKEND_DIR}/target") {
                    bat "dir"
                    bat "java -jar ${JAR_NAME} --server.port=${BACKEND_PORT}"
                }
            }
        }
    }

    post {
        always {
            echo "Pipeline finished"
        }
    }
}
