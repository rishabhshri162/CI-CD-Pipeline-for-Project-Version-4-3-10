pipeline {
    agent any

    environment {
        JAVA_HOME = "C:\\Program Files\\Java\\jdk-11.0.15.1"
        PATH = "${JAVA_HOME}\\bin;${PATH}"
        TOMCAT_HOME = 'C:\\Users\\risha\\Downloads\\apache-tomcat-9.0.109-windows-x64\\apache-tomcat-9.0.109'
        DOCKER_PATH = 'C:\\Users\\risha\\Desktop\\Project3 Docker'
        TOMCAT_WEBAPPS_PATH = "C:\\Users\\risha\\Downloads\\apache-tomcat-9.0.109-windows-x64\\apache-tomcat-9.0.109\\webapps"
        TOMCAT_SERVICE_NAME = 'Tomcat9'
        MAVEN_HOME = "C:\\ProgramData\\Jenkins\\.jenkins\\tools\\hudson.tasks.Maven_MavenInstallation\\Maven"

        APP_URL = "http://localhost:8080/project_3"
    }

    stages {

        stage('Cleanup Workspace') {
            steps {
                deleteDir()
            }
        }

        stage('Clone Repository') {
            steps {
                git url: 'https://github.com/rishabhshri162/Test-project3.git', branch: 'main'
            }
        }

        stage('Build WAR File') {
            steps {
                // 🔧 FIX: pom.xml is at ROOT
                bat "\"%MAVEN_HOME%\\bin\\mvn.cmd\" clean package -DskipTests"
            }
        }

        stage('Copy WAR to Tomcat') {
            steps {
                script {
                    // 🔧 FIX: project_3 folder does NOT exist in workspace
                    bat "rmdir /S /Q \"${TOMCAT_WEBAPPS_PATH}\\project_3\" || exit 0"
                    bat "del /Q \"${TOMCAT_WEBAPPS_PATH}\\project_3.war\" || exit 0"

                    // 🔧 FIX: WAR is directly under target/
                    bat "copy /Y target\\project_3.war \"${TOMCAT_WEBAPPS_PATH}\\project_3.war\""
                    bat "copy /Y target\\project_3.war \"${DOCKER_PATH}\\project_3.war\""
                }
            }
        }

        stage('Ensure Tomcat Service Exists') {
            steps {
                script {
                    def exists = bat(script: "sc query ${TOMCAT_SERVICE_NAME}", returnStatus: true)
                    if (exists != 0) {
                        bat "cd /d ${TOMCAT_HOME}\\bin && service.bat install"
                    }
                }
            }
        }

        stage('Restart Tomcat') {
            steps {
                script {
                    bat "sc stop ${TOMCAT_SERVICE_NAME} || exit 0"
                    sleep time: 5, unit: 'SECONDS'
                    bat "sc start ${TOMCAT_SERVICE_NAME}"
                    sleep time: 20, unit: 'SECONDS'
                }
            }
        }

        stage('Verify Deployment (Retry Enabled)') {
            steps {
                retry(3) {
                    bat '''
powershell -NoProfile -Command "try { Invoke-WebRequest -Uri http://localhost:8080/project_3 -UseBasicParsing -TimeoutSec 10 | Out-Null; Write-Host 'Application is UP' } catch { Write-Host 'Application NOT reachable'; exit 1 }"
'''
                }
            }
        }
    }

    post {
        always {
            deleteDir()
        }
        success {
            echo "🎉 DEPLOYMENT SUCCESSFUL"
        }
        failure {
            echo "❌ DEPLOYMENT FAILED"
        }
    }
}
